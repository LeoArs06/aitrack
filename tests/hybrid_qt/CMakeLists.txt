cmake_minimum_required(VERSION 3.16)
project(AITrackHybridQtTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_SUPPRESS_REGENERATION ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core SerialPort Network Widgets)
set(CLIENT "${CMAKE_CURRENT_LIST_DIR}/../../Client/src")
file(GLOB CORE "${CLIENT}/hybrid/*.cpp")
add_library(hybrid_qt_adapter STATIC
    ${CORE} "${CLIENT}/hybrid_qt/ImuWorker.cpp" "${CLIENT}/hybrid_qt/ImuWorker.h"
    "${CLIENT}/model/UDPSender.cpp" "${CLIENT}/model/Config.cpp")
target_compile_definitions(hybrid_qt_adapter PUBLIC NOMINMAX)
if(MSVC)
    target_compile_options(hybrid_qt_adapter PUBLIC /EHsc)
endif()
target_link_libraries(hybrid_qt_adapter PUBLIC Qt5::Core Qt5::SerialPort ws2_32)
add_executable(hybrid_qt_tests hybrid_qt_tests.cpp)
target_link_libraries(hybrid_qt_tests PRIVATE hybrid_qt_adapter Qt5::Network)
add_executable(calibration_qt_tests calibration_qt_tests.cpp)
target_link_libraries(calibration_qt_tests PRIVATE hybrid_qt_adapter)
add_executable(serial_retry_qt_tests serial_retry_qt_tests.cpp)
target_link_libraries(serial_retry_qt_tests PRIVATE hybrid_qt_adapter)
qt5_wrap_ui(UI_HEADER "${CLIENT}/view/MainWindow.ui")
add_executable(hybrid_ui_tests hybrid_ui_tests.cpp ${UI_HEADER})
target_include_directories(hybrid_ui_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CLIENT}/..")
target_link_libraries(hybrid_ui_tests PRIVATE Qt5::Widgets Qt5::SerialPort)
